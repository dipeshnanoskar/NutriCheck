import 'dart:convert';
import 'package:http/http.dart' as http;
import 'package:flutter_secure_storage/flutter_secure_storage.dart';
import 'constants.dart';

class ApiException implements Exception {
  final String message;
  final int? statusCode;
  final Map<String, dynamic>? errors;
  ApiException(this.message, {this.statusCode, this.errors});

  @override
  String toString() => message;
}

/// Thin wrapper around http package that injects the bearer token,
/// retries once on a 401 using the refresh token, and decodes JSON.
class ApiClient {
  ApiClient._internal();
  static final ApiClient instance = ApiClient._internal();

  final _storage = const FlutterSecureStorage();
  final _client = http.Client();

  Future<String?> get _accessToken => _storage.read(key: StorageKeys.accessToken);
  Future<String?> get _refreshToken => _storage.read(key: StorageKeys.refreshToken);

  Future<void> saveTokens({required String access, required String refresh}) async {
    await _storage.write(key: StorageKeys.accessToken, value: access);
    await _storage.write(key: StorageKeys.refreshToken, value: refresh);
  }

  Future<void> clearTokens() async {
    await _storage.delete(key: StorageKeys.accessToken);
    await _storage.delete(key: StorageKeys.refreshToken);
  }

  Future<bool> get isLoggedIn async => (await _accessToken) != null;

  Uri _uri(String path, [Map<String, dynamic>? query]) {
    final base = Uri.parse(ApiConfig.baseUrl + path);
    if (query == null) return base;
    return base.replace(queryParameters: {
      ...base.queryParameters,
      ...query.map((k, v) => MapEntry(k, v.toString())),
    });
  }

  Future<Map<String, String>> _headers({bool auth = true}) async {
    final headers = {'Content-Type': 'application/json'};
    if (auth) {
      final token = await _accessToken;
      if (token != null) headers['Authorization'] = 'Bearer $token';
    }
    return headers;
  }

  Future<bool> _tryRefresh() async {
    final refresh = await _refreshToken;
    if (refresh == null) return false;
    final resp = await _client.post(
      _uri('/auth/refresh'),
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer $refresh',
      },
    );
    if (resp.statusCode == 200) {
      final body = jsonDecode(resp.body);
      await _storage.write(key: StorageKeys.accessToken, value: body['access_token']);
      return true;
    }
    return false;
  }

  dynamic _decode(http.Response resp) {
    if (resp.body.isEmpty) return null;
    try {
      return jsonDecode(resp.body);
    } catch (_) {
      return null;
    }
  }

  Future<dynamic> _handle(http.Response resp) async {
    if (resp.statusCode >= 200 && resp.statusCode < 300) {
      return _decode(resp);
    }
    final body = _decode(resp);
    throw ApiException(
      body?['error'] ?? body?['errors']?.toString() ?? 'Request failed (${resp.statusCode})',
      statusCode: resp.statusCode,
      errors: body?['errors'] is Map ? Map<String, dynamic>.from(body['errors']) : null,
    );
  }

  Future<dynamic> get(String path, {Map<String, dynamic>? query, bool auth = true}) async {
    var resp = await _client
        .get(_uri(path, query), headers: await _headers(auth: auth))
        .timeout(ApiConfig.requestTimeout);
    if (resp.statusCode == 401 && auth && await _tryRefresh()) {
      resp = await _client
          .get(_uri(path, query), headers: await _headers(auth: auth))
          .timeout(ApiConfig.requestTimeout);
    }
    return _handle(resp);
  }

  Future<dynamic> post(String path, {Object? body, bool auth = true}) async {
    var resp = await _client
        .post(_uri(path), headers: await _headers(auth: auth), body: jsonEncode(body ?? {}))
        .timeout(ApiConfig.requestTimeout);
    if (resp.statusCode == 401 && auth && await _tryRefresh()) {
      resp = await _client
          .post(_uri(path), headers: await _headers(auth: auth), body: jsonEncode(body ?? {}))
          .timeout(ApiConfig.requestTimeout);
    }
    return _handle(resp);
  }

  Future<dynamic> put(String path, {Object? body, bool auth = true}) async {
    final resp = await _client
        .put(_uri(path), headers: await _headers(auth: auth), body: jsonEncode(body ?? {}))
        .timeout(ApiConfig.requestTimeout);
    return _handle(resp);
  }

  Future<dynamic> delete(String path, {bool auth = true}) async {
    final resp = await _client
        .delete(_uri(path), headers: await _headers(auth: auth))
        .timeout(ApiConfig.requestTimeout);
    return _handle(resp);
  }

  /// Multipart upload for OCR image analysis.
  Future<dynamic> postMultipart(
    String path, {
    required List<int> fileBytes,
    required String filename,
    Map<String, String>? fields,
  }) async {
    final token = await _accessToken;
    final request = http.MultipartRequest('POST', _uri(path));
    if (token != null) request.headers['Authorization'] = 'Bearer $token';
    if (fields != null) request.fields.addAll(fields);
    request.files.add(http.MultipartFile.fromBytes('image', fileBytes, filename: filename));

    final streamed = await _client.send(request).timeout(const Duration(seconds: 30));
    final resp = await http.Response.fromStream(streamed);
    return _handle(resp);
  }

  /// Multipart upload for the two-photo OCR flow (ingredients + nutrition
  /// table). [files] maps the multipart field name (matching what the
  /// Flask route expects, e.g. 'ingredients_image', 'nutrition_image')
  /// to that file's raw bytes and filename.
  Future<dynamic> postMultipartFiles(
    String path, {
    required Map<String, ({List<int> bytes, String filename})> files,
    Map<String, String>? fields,
  }) async {
    final token = await _accessToken;
    final request = http.MultipartRequest('POST', _uri(path));
    if (token != null) request.headers['Authorization'] = 'Bearer $token';
    if (fields != null) request.fields.addAll(fields);
    for (final entry in files.entries) {
      request.files.add(http.MultipartFile.fromBytes(
        entry.key,
        entry.value.bytes,
        filename: entry.value.filename,
      ));
    }

    final streamed = await _client.send(request).timeout(const Duration(seconds: 30));
    final resp = await http.Response.fromStream(streamed);
    return _handle(resp);
  }
}

/// Status colors for suitability badges (suitable / caution / avoid).
class StatusStyle {
  final Color bg;
  final Color fg;
  const StatusStyle(this.bg, this.fg);

  static StatusStyle forStatus(String status) {
    switch (status.toLowerCase()) {
      case 'suitable':
      case 'generally suitable':
        return const StatusStyle(AppColors.greenLight, AppColors.greenPrimary);
      case 'caution':
      case 'use caution':
        return const StatusStyle(AppColors.amberLight, AppColors.amber);
      case 'limit':
        return const StatusStyle(AppColors.orangeLight, AppColors.orange);
      case 'avoid':
        return const StatusStyle(AppColors.redLight, AppColors.red);
      default:
        return const StatusStyle(AppColors.border, AppColors.muted);
    }
  }
}

import 'package:flutter/foundation.dart';
import '../core/api_client.dart';
import '../models/user_profile.dart';
import '../services/auth_service.dart';

enum AuthStatus { unknown, authenticated, unauthenticated }

class AuthProvider extends ChangeNotifier {
  final _authService = AuthService();

  AuthStatus status = AuthStatus.unknown;
  UserProfile? profile;
  String? errorMessage;
  bool isLoading = false;

  Future<void> bootstrap() async {
    final loggedIn = await ApiClient.instance.isLoggedIn;
    if (!loggedIn) {
      status = AuthStatus.unauthenticated;
      notifyListeners();
      return;
    }
    try {
      profile = await _authService.fetchMe();
      status = AuthStatus.authenticated;
    } catch (_) {
      status = AuthStatus.unauthenticated;
    }
    notifyListeners();
  }

  Future<bool> register({
    required String name,
    required String email,
    required String password,
    required int age,
  }) => _run(() async {
        profile = await _authService.register(name: name, email: email, password: password, age: age);
        status = AuthStatus.authenticated;
      });

  Future<bool> login({required String email, required String password}) => _run(() async {
        profile = await _authService.login(email: email, password: password);
        status = AuthStatus.authenticated;
      });

  Future<bool> updateProfile({String? name, int? age}) => _run(() async {
        profile = await _authService.updateProfile(name: name, age: age);
      });

  Future<void> logout() async {
    await _authService.logout();
    profile = null;
    status = AuthStatus.unauthenticated;
    notifyListeners();
  }

  Future<bool> _run(Future<void> Function() action) async {
    isLoading = true;
    errorMessage = null;
    notifyListeners();
    try {
      await action();
      isLoading = false;
      notifyListeners();
      return true;
    } catch (e) {
      isLoading = false;
      errorMessage = e.toString();
      notifyListeners();
      return false;
    }
  }
}

class HistoryProvider extends ChangeNotifier {
  final _service = HistoryService();

  List<HistoryItem> items = [];
  bool isLoading = false;
  String? errorMessage;

  Future<void> load() async {
    isLoading = true;
    errorMessage = null;
    notifyListeners();
    try {
      items = await _service.list();
    } catch (e) {
      errorMessage = e.toString();
    }
    isLoading = false;
    notifyListeners();
  }

  Future<void> delete(String id) async {
    await _service.delete(id);
    items.removeWhere((i) => i.id == id);
    notifyListeners();
  }

  Future<void> clearAll() async {
    await _service.clearAll();
    items = [];
    notifyListeners();
  }
}

enum ScanStage {
  idle,
  lookingUpProduct,
  notFound,
  analyzing,
  result,
  error,
  awaitingNutritionPhoto, // NEW - ingredients captured, prompting for nutrition photo
}

/// Drives the barcode-scan -> product-detail -> analyzing -> result flow,
/// and the OCR-camera -> review -> analyzing -> result flow.
class ScanProvider extends ChangeNotifier {
  final _productService = ProductService();
  final _analysisService = AnalysisService();

  ScanStage stage = ScanStage.idle;
  String category = 'food'; // 'food' | 'drinks'
  Product? product;
  AnalysisResult? result;
  String? ocrIngredientsText;
  String? errorMessage;
  String? lastBarcode;

  // NEW - holds the ingredients photo while the user is prompted to also
  // capture the nutrition table. Cleared once the two-photo analysis
  // completes (success or failure) or the flow is reset.
  Uint8List? pendingIngredientsImage;

  // NEW - when the OCR flow is started from a known product (e.g. a
  // barcode-resolved product with missing ingredients/nutrition), this
  // carries the product's real name through to the OCR result/history
  // instead of the generic "Scanned Label" fallback.
  String? productNameOverride;

  void setCategory(String value) {
    category = value;
    notifyListeners();
  }

  void reset() {
    stage = ScanStage.idle;
    product = null;
    result = null;
    ocrIngredientsText = null;
    errorMessage = null;
    pendingIngredientsImage = null;
    productNameOverride = null;
    notifyListeners();
  }

  Future<void> lookupBarcode(String barcode) async {
    lastBarcode = barcode;
    stage = ScanStage.lookingUpProduct;
    notifyListeners();
    try {
      product = await _productService.lookupBarcode(barcode);
      stage = ScanStage.idle; // caller navigates to product-detail screen
      notifyListeners();
    } on ProductNotFoundException {
      stage = ScanStage.notFound;
      notifyListeners();
    } catch (e) {
      errorMessage = e.toString();
      stage = ScanStage.error;
      notifyListeners();
    }
  }

  Future<void> analyzeCurrentProduct() async {
    if (product == null) return;
    stage = ScanStage.analyzing;
    notifyListeners();
    try {
      result = await _analysisService.analyzeProduct(product!, category: category);
      stage = ScanStage.result;
    } catch (e) {
      errorMessage = e.toString();
      stage = ScanStage.error;
    }
    notifyListeners();
  }

  Future<void> analyzeOcrImage(Uint8List imageBytes) async {
    stage = ScanStage.analyzing;
    notifyListeners();
    try {
      final outcome = await _analysisService.analyzeImage(
        imageBytes,
        category: category,
        productName: productNameOverride ?? 'Scanned Label',
      );
      result = outcome.result;
      ocrIngredientsText = outcome.ingredientsText;
      stage = ScanStage.result;
    } catch (e) {
      errorMessage = e.toString();
      stage = ScanStage.error;
    }
    notifyListeners();
  }

  /// NEW - Step 1 of the two-photo flow: store the reviewed ingredients
  /// photo and move to the "now scan nutrition" prompt, without calling
  /// the backend yet.
  void submitIngredientsPhoto(Uint8List imageBytes) {
    pendingIngredientsImage = imageBytes;
    stage = ScanStage.awaitingNutritionPhoto;
    notifyListeners();
  }

  /// NEW - Step 2 of the two-photo flow: send both photos together.
  /// [nutritionImageBytes] is optional - if the user skips it, analysis
  /// proceeds with ingredients only.
  Future<void> analyzeTwoImages({Uint8List? nutritionImageBytes}) async {
    if (pendingIngredientsImage == null) return;
    stage = ScanStage.analyzing;
    notifyListeners();
    try {
          final outcome = await _analysisService.analyzeTwoImages(
        pendingIngredientsImage!,
        nutritionImageBytes: nutritionImageBytes,
        category: category,
        productName: productNameOverride ?? 'Scanned Label',
      );
      result = outcome.result;
      ocrIngredientsText = outcome.ingredientsText;
      stage = ScanStage.result;
    } catch (e) {
      errorMessage = e.toString();
      stage = ScanStage.error;
    }
    pendingIngredientsImage = null;
    notifyListeners();
  }
}

      if (file == null) {
        setState(() => _busy = false);

  /// Splits a raw ingredients string into individual ingredient lines.
  /// Splits on top-level commas/semicolons only - text inside
  /// parentheses or brackets (e.g. "Vegetable oil (palm, sunflower)")
  /// is kept together as part of that ingredient's description.
  static List<String> _splitIngredients(String rawText) {
    // European ingredient lists often use a comma as a DECIMAL POINT
    // (e.g. "Cacao maigre 7,4%"). Protect digit-comma-digit patterns
    // before splitting so "7,4" isn't mistaken for a list separator.
    final text = rawText.replaceAllMapped(
      RegExp(r'(\d),(\d)'),
      (m) => '${m.group(1)}\u0001${m.group(2)}',
    );

    final List<String> result = [];
    int depth = 0;
    final buffer = StringBuffer();

    for (final rune in text.runes) {
      final char = String.fromCharCode(rune);
      if (char == '(' || char == '[') {
        depth++;
        buffer.write(char);
      } else if (char == ')' || char == ']') {
        depth--;
        buffer.write(char);
      } else if ((char == ',' || char == ';') && depth <= 0) {
        final piece = buffer.toString().trim();
        if (piece.isNotEmpty) result.add(piece);
        buffer.clear();
      } else {
        buffer.write(char);
      }
    }
    final last = buffer.toString().trim();
    if (last.isNotEmpty) result.add(last);

    // Restore protected decimal commas and capitalize each line.
    return result.map((e) {
      final restored = e.replaceAll('\u0001', ',');
      if (restored.isEmpty) return restored;
      return restored[0].toUpperCase() + restored.substring(1);
    }).toList();
  }

  /// True when every nutrition value is effectively zero/blank - a strong
  /// signal that Open Food Facts had no real nutrition data for this
  /// product, rather than the product genuinely having all-zero values.
  static bool _isNutritionEmpty(dynamic nutrition) {
    for (final entry in nutrition.entries) {
      final digits = RegExp(r'[\d.]+').firstMatch(entry.value.toString());
      final value = digits != null ? double.tryParse(digits.group(0)!) ?? 0 : 0;
      if (value > 0) return false;
    }
    return true;
  }

/// Splits raw OCR ingredients text into individual items for bullet
/// display. Splits on commas, but keeps bracketed/parenthetical groups
/// together - e.g. "RAISING AGENTS [INS 503(ii), INS 500(ii)]" stays one
/// item instead of being split into three.
List<String> _splitIngredientItems(String text) {
  final items = <String>[];
  final buffer = StringBuffer();
  var depth = 0;

  for (final char in text.split('')) {
    if (char == '(' || char == '[') {
      depth++;
      buffer.write(char);
    } else if (char == ')' || char == ']') {
      depth = depth > 0 ? depth - 1 : 0;
      buffer.write(char);
    } else if (char == ',' && depth == 0) {
      final item = buffer.toString().trim();
      if (item.isNotEmpty) items.add(item);
      buffer.clear();
    } else {
      buffer.write(char);
    }
  }
  final last = buffer.toString().trim();
  if (last.isNotEmpty) items.add(last);

  return items;
}

class _ScanScreenState extends State<ScanScreen> {
  final MobileScannerController _controller = MobileScannerController(
    detectionSpeed: DetectionSpeed.noDuplicates,
    formats: const [BarcodeFormat.ean13, BarcodeFormat.ean8, BarcodeFormat.upcA, BarcodeFormat.upcE],
  );
  final TextEditingController _manualBarcodeController = TextEditingController();
  final FocusNode _manualBarcodeFocus = FocusNode();

  bool _handling = false;
  bool _torchOn = false;
  String? _manualError;

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addPostFrameCallback((_) {
      context.read<ScanProvider>().setCategory(widget.initialCategory);
    });
  }

  @override
  void dispose() {
    _controller.dispose();
    _manualBarcodeController.dispose();
    _manualBarcodeFocus.dispose();
    super.dispose();
  }

  Future<void> _onDetect(BarcodeCapture capture) async {
    if (_handling) return;
    final barcode = capture.barcodes.firstOrNull?.rawValue;
    if (barcode == null || barcode.isEmpty) return;
    await _controller.stop();
    await _handleBarcode(barcode);
    if (!mounted) return;
    await _controller.start();
  }

  Future<void> _handleManualSubmit() async {
    final value = _manualBarcodeController.text.trim();
    if (value.isEmpty) {
      setState(() => _manualError = 'Enter a barcode number');
      return;
    }
    if (!RegExp(r'^\d{6,14}$').hasMatch(value)) {
      setState(() => _manualError = 'Enter a valid numeric barcode');
      return;
    }
    setState(() => _manualError = null);
    _manualBarcodeFocus.unfocus();
    await _controller.stop();
    await _handleBarcode(value);
    if (!mounted) return;
    await _controller.start();
  }

  /// Shared resolution path for both camera-detected and manually-entered
  /// barcodes. Uses the existing ScanProvider -> ProductService -> backend
  /// barcode-analysis API - nothing about that flow changes.
  Future<void> _handleBarcode(String barcode) async {
    if (_handling) return;
    setState(() => _handling = true);

    final scan = context.read<ScanProvider>();
    await scan.lookupBarcode(barcode);

    if (!mounted) {
      setState(() => _handling = false);
      return;
    }

    if (scan.stage == ScanStage.notFound) {
      await Navigator.of(context).push(
        MaterialPageRoute(builder: (_) => NotFoundScreen(barcode: barcode)),
      );
    } else if (scan.product != null) {
      await Navigator.of(context).push(
        MaterialPageRoute(builder: (_) => const ProductDetailScreen()),
      );
    } else if (scan.stage == ScanStage.error) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(scan.errorMessage ?? 'Something went wrong')),
      );
    }

    if (!mounted) return;
    setState(() => _handling = false);
  }

class OcrAnalysisOutcome {
  final AnalysisResult result;
  final String ingredientsText;
  OcrAnalysisOutcome({required this.result, required this.ingredientsText});
}

class AnalysisService {
  final _api = ApiClient.instance;

  Future<AnalysisResult> analyzeProduct(Product product, {String category = 'food'}) async {
    final data = await _api.post('/analyze/product', body: {
      'name': product.name,
      'brand': product.brand,
      'image': product.image,
      'barcode': product.barcode,
      'ingredients': product.ingredients,
      'nutrition': product.nutrition.toJson(),
      'category': category,
    });
    return AnalysisResult.fromJson(data['result']);
  }
  Future<OcrAnalysisOutcome> analyzeImage(
    Uint8List imageBytes, {
    String filename = 'label.jpg',
    String category = 'food',
    String productName = 'Scanned Label',
  }) async {
    final data = await _api.postMultipart(
      '/analyze/ocr',
      fileBytes: imageBytes,
      filename: filename,
      fields: {'category': category, 'name': productName},
    );
    return OcrAnalysisOutcome(
      result: AnalysisResult.fromJson(data['result']),
      ingredientsText: data['ingredients_text'] ?? '',
    );
  }

  /// Two-photo OCR flow: ingredients label + nutrition table.
  /// [nutritionImageBytes] is optional - if null, analysis proceeds with
  /// ingredients only (matches the backend's optional nutrition_image).
  Future<TwoPhotoOcrOutcome> analyzeTwoImages(
    Uint8List ingredientsImageBytes, {
    Uint8List? nutritionImageBytes,
    String category = 'food',
    String productName = 'Scanned Label',
  }) async {
    final files = {
      'ingredients_image': (bytes: ingredientsImageBytes, filename: 'ingredients.jpg'),
      if (nutritionImageBytes != null)
        'nutrition_image': (bytes: nutritionImageBytes, filename: 'nutrition.jpg'),
    };
    final data = await _api.postMultipartFiles(
      '/analyze/ocr-full',
      files: files,
      fields: {'category': category, 'name': productName},
    );
    return TwoPhotoOcrOutcome(
      result: AnalysisResult.fromJson(data['result']),
      ingredientsText: data['ingredients_text'] ?? '',
    );
  }
}

class TwoPhotoOcrOutcome {
  final AnalysisResult result;
  final String ingredientsText;
  TwoPhotoOcrOutcome({required this.result, required this.ingredientsText});
}

class AuthService {

  Future<UserProfile> register({
    required String name,
    required String email,
    required String password,
    required int age,
  }) async {
    final data = await _api.post(
      '/auth/register',
      body: {'name': name, 'email': email, 'password': password, 'age': age},
      auth: false,
    );
    await _api.saveTokens(access: data['access_token'], refresh: data['refresh_token']);
    return UserProfile.fromJson(data['user']);
  }

  Future<UserProfile> login({required String email, required String password}) async {
    final data = await _api.post(
      '/auth/login',
      body: {'email': email, 'password': password},
      auth: false,
    );
    await _api.saveTokens(access: data['access_token'], refresh: data['refresh_token']);
    return UserProfile.fromJson(data['user']);
  }

  Future<UserProfile> fetchMe() async {
    final data = await _api.get('/auth/me');
    return UserProfile.fromJson(data['user']);
  }

  Future<UserProfile> updateProfile({String? name, int? age}) async {
    final body = <String, dynamic>{};
    if (name != null) body['name'] = name;
    if (age != null) body['age'] = age;
    final data = await _api.put('/auth/me', body: body);
    return UserProfile.fromJson(data['user']);
  }

  Future<void> logout() async {
    await _api.clearTokens();
  }

  Future<bool> get isLoggedIn => _api.isLoggedIn;
}

}

class ProductService {
  final _api = ApiClient.instance;

  Future<Product> lookupBarcode(String barcode) async {
    try {
      final data = await _api.get('/products/barcode/$barcode');
      return Product.fromJson(data['product']);
    } on ApiException catch (e) {
      if (e.statusCode == 404) throw ProductNotFoundException(barcode);
      rethrow;
    }
  }

  Future<List<Map<String, dynamic>>> search(String query) async {
    final data = await _api.get('/products/search', query: {'q': query});
    return List<Map<String, dynamic>>.from(data['results'] ?? []);
  }
}

import 'package:flutter/material.dart';

import '../models/history_item.dart';

class HistoryCard extends StatelessWidget {
  final HistoryItem item;
  final VoidCallback? onTap;
  final VoidCallback? onDismiss;

  const HistoryCard({super.key, required this.item, this.onTap, this.onDismiss});

  @override
  Widget build(BuildContext context) {
    final card = Container(
      margin: const EdgeInsets.only(bottom: 12),
      padding: const EdgeInsets.all(12),
      decoration: BoxDecoration(
        color: AppColors.surface,
        borderRadius: BorderRadius.circular(kCardRadius),
        border: Border.all(color: AppColors.border),
      ),
      child: Row(
        children: [
          ClipRRect(
            borderRadius: BorderRadius.circular(10),
            child: item.image.isNotEmpty
                ? Image.network(item.image, width: 52, height: 52, fit: BoxFit.cover,
                    errorBuilder: (_, __, ___) => _fallbackThumb())
                : _fallbackThumb(),
          ),
          const SizedBox(width: 12),
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text(item.productName,
                    maxLines: 1,
                    overflow: TextOverflow.ellipsis,
                    style: const TextStyle(fontWeight: FontWeight.w600, color: AppColors.ink)),
                if (item.brand.isNotEmpty)
                  Text(item.brand,
                      maxLines: 1,
                      overflow: TextOverflow.ellipsis,
                      style: const TextStyle(fontSize: 12, color: AppColors.muted)),
              ],
            ),
          ),
          const SizedBox(width: 8),
          StatusBadge(status: item.overall, fontSize: 11),
        ],
      ),
    );

    final tappable = InkWell(borderRadius: BorderRadius.circular(kCardRadius), onTap: onTap, child: card);

    if (onDismiss == null) return tappable;

    return Dismissible(
      key: ValueKey(item.id),
      direction: DismissDirection.endToStart,
      background: Container(
        margin: const EdgeInsets.only(bottom: 12),
        decoration: BoxDecoration(color: AppColors.redLight, borderRadius: BorderRadius.circular(kCardRadius)),
        alignment: Alignment.centerRight,
        padding: const EdgeInsets.only(right: 20),
        child: const Icon(Icons.delete_outline, color: AppColors.red),
      ),
      onDismissed: (_) => onDismiss!(),
      child: tappable,
    );
  }

  Widget _fallbackThumb() => Container(
        width: 52,
        height: 52,
        color: AppColors.greenLight,
        child: const Icon(Icons.fastfood_rounded, color: AppColors.greenPrimary, size: 22),
      );
}
